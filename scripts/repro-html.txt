    var NOTE_SEMITONE = {
        'C':0,'C#':1,'Db':1,'D':2,'D#':3,'Eb':3,'E':4,'F':5,'F#':6,'Gb':6,
        'G':7,'G#':8,'Ab':8,'A':9,'A#':10,'Bb':10,'B':11
    };

    function noteToFrequency(noteName) {
        var match = noteName.match(/^([A-G]#?b?)(\d+)$/);
        if (!match) return 440;
        var note = match[1];
        var octave = parseInt(match[2], 10);
        var semi = NOTE_SEMITONE[note];
        if (semi === undefined) return 440;
        var midi = (octave + 1) * 12 + semi;
        return 440 * Math.pow(2, (midi - 69) / 12);
    }

    function getAudioContext() {
        if (!audioCtx) {
            try { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); }
            catch(e) { return null; }
        }
        if (audioCtx.state === 'suspended') audioCtx.resume();
        return audioCtx;
    }

    function playNote(frequency, duration, startTime) {
        var ctx = getAudioContext();
        if (!ctx) return;
        var now = ctx.currentTime;
        var osc = ctx.createOscillator();
        var gain = ctx.createGain();
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(frequency, now + startTime);
        gain.gain.setValueAtTime(0, now + startTime);
        gain.gain.linearRampToValueAtTime(0.6, now + startTime + 0.02);
        gain.gain.setValueAtTime(0.6, now + startTime + duration - 0.1);
        gain.gain.linearRampToValueAtTime(0, now + startTime + duration);
        osc.connect(gain);
        gain.connect(ctx.destination);
        osc.start(now + startTime);
        osc.stop(now + startTime + duration + 0.05);
    }

    function playChord() {
        if (!activeChord || !CHORD_DB[activeChord]) return;
        var notes = CHORD_DB[activeChord];
        var duration = 1.5;
        var btn = document.getElementById('playChordBtn');
        btn.classList.add('playing');
        btn.textContent = '▶ Tocando...';
        notes.forEach(function(n) {
            var freq = noteToFrequency(n);
            if (freq) playNote(freq, duration, 0);
        });
        setTimeout(function() {
            btn.classList.remove('playing');
            btn.textContent = '▶ Tocar Acorde';
        }, duration * 1000 + 100);
    }

    function renderKeyboard() {

            div.style.left = offset + 'px';
            div.style.top = '0';
            container.appendChild(div);
        });
        container.style.width = (whiteNotes.length * kw) + 'px';
    }

        var prompt = 'Você é um professor de teclado. Gere uma cifra para "' + query + '". Responda com JSON: { "title", "artist", "key", "chords", "difficulty", "tips", "lyrics" }';
        AppacadabraAI.withSchema({
            type: 'object',
            properties: {
                title: { type: 'string' }, artist: { type: 'string' }, key: { type: 'string' },
                chords: { type: 'array', items: { type: 'string' } },
                difficulty: { type: 'number' }, tips: { type: 'string' }, lyrics: { type: 'string' }
            },
            required: ['title','artist','key','chords','difficulty','tips','lyrics']
        }).generate(prompt, 'handleAISongResult');
    }

    window.handleAISongResult = function(success, data) {
        if (!success || !data || !data.chords) return;
        songs.push({
            id: generateId(), title: data.title, artist: data.artist, key: data.key,
            chords: data.chords, difficulty: data.difficulty, tips: data.tips,
            lyrics: data.lyrics, isUserAdded: true
        });
        localStorage.setItem('tecla_facil_songs', JSON.stringify(songs));
        renderSongList();
    };

    function initApp() {
        var storedSongs = localStorage.getItem('tecla_facil_songs');
        if (storedSongs) { try { songs = JSON.parse(storedSongs); } catch(e) { songs = []; } }
        if (songs.length === 0) songs = DEFAULT_SONGS.slice();
        localStorage.setItem('tecla_facil_songs', JSON.stringify(songs));
        var storedLog = localStorage.getItem('tecla_facil_practice_log');
        if (storedLog) { try { practiceLog = JSON.parse(storedLog); } catch(e) {} }
        initChordDB();
        renderKeyboard();
        renderSongList();
    }

    document.addEventListener('DOMContentLoaded', initApp);
